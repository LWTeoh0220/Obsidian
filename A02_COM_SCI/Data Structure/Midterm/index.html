def analyze_server_log(file_path): stack = [] try: with open(file_path, 'r', encoding='utf-8') as f: lines = f.readlines() for line_num, line in enumerate(lines, 1): line = line.strip() if not line: continue # 將 "[START] api_name" 切割成 action 和 api_name parts = line.split(" ", 1) if len(parts) != 2: continue action, api_name = parts[0], parts[1] # 遇到 [START] 就 push 進 Stack if action == "[START]": stack.append(api_name) # 遇到 [END] 就檢查 Stack 最上方 elif action == "[END]": # 防呆：如果 Stack 已經空了卻還有 END if not stack: print(f"❌ 行號 {line_num} 異常：遇到 [END] {api_name}，但根本沒有 START 過！") return # Pop 出最上面的 API top_api = stack.pop() # 檢查是否匹配 if top_api != api_name: print(f"⚠️ 發現異常！發生在第 {line_num} 行：") print(f"👉 系統預期要結束的 API 是：【{top_api}】 (因為它最後執行)") print(f"👉 但實際卻提早結束了：【{api_name}】") print(f"🎯 結論：卡住/發生 Deadlock 的 API 是 【{top_api}】") return # 題目說找出「第一個」異常，所以找到就可以結束了 # 如果跑完日誌，Stack 裡面還有東西，代表有人沒 END if stack: print(f"⚠️ 日誌結束，但有 API 尚未完成 (缺少 END)：{stack}") else: print("✅ 伺服器日誌一切正常，沒有異常。") except FileNotFoundError: print(f"找不到檔案：{file_path}，請確認檔名與路徑。") # 執行分析 analyze_server_log("Q7_server_log.txt")
